        long callback = pending.getCallbackAfterSeconds();
        if (callback < 0) {
            callback = 0;
        }
        queueDAO.remove(Nullability.castToNonnull(QueueUtils.getQueueName(pending)), pending.getTaskId());

        if (callback < 0) {
            callback = 0;
        }

                pending.getTaskId(),
                pending.getWorkflowPriority(),
